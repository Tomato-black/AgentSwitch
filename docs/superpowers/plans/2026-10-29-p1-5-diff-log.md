```go
type DocDiffLine struct {
	Op   string // "+" | "-" | "~"
	Path string
	Old, New string
}

func DocDiff(oldJSON, newJSON string) []DocDiffLine {
	oldMap, newMap := flattenJSON(oldJSON), flattenJSON(newJSON)
	keys := union sorted
	for _, k := range keys {
		ov, okO := oldMap[k]; nv, okN := newMap[k]
		switch {
		case !okO: append "+", mask(nv)
		case !okN: append "-", mask(ov)
		case ov != nv: append "~", mask(ov), mask(nv)
		}
	}
	return lines
}
```

`flattenJSON` 解析对象并递归展开为 `a.b.c -> JSON 值文本`；解析失败返回空 map。
`mask` 对路径含 TOKEN/KEY/SECRET/AUTH/PASSWORD 返回值 `***`。

var openPath = func(path string) error {
	var cmd *exec.Cmd
	switch runtime.GOOS {
	case "darwin": cmd = exec.Command("open", path)
	case "windows": cmd = exec.Command("explorer", path)
	default: cmd = exec.Command("xdg-open", path)
	}
	return cmd.Start()
}

func (g Gateway) OpenLogDir() error {
	p := g.A.LogPath()
	if p == "" { return fmt.Errorf("日志路径未配置") }
	return openPath(filepath.Dir(p))
}
