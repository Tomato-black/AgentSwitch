# P1-4 通知/告警实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** 全渠道冷却、连续故障转移、上游额度耗尽时给出应用内提示 + 系统通知；去重/限频，设置页可关闭。

**Architecture:** bridge 读写 `alerts.enabled`（默认开）；desktop 新增纯逻辑 `alertState.evaluate(rec, now)`，在 `OnRecord` 中调用；命中时设置侧栏告警文案并调用 `mygo.NewNotification`（不支持时仅应用内）。不改 gateway/协议。

**Tech Stack:** Go 1.27、MyGo Notification；测试离线。

**Spec:** `docs/product/roadmap.md` P1-4；`docs/product/feature-gap-analysis.md` A7。

## Global Constraints

- 不自动 commit。
- 告警内容不含密钥、渠道 Key、prompt/正文。
- 去重/限频：全冷却/连续切换 5 分钟，额度耗尽按渠道 10 分钟。
- 系统通知不可用时降级为应用内侧栏提示，不阻塞请求。
- 默认开启；设置页可关闭。

```go
const alertsEnabledKey = "alerts.enabled"

func (g Gateway) AlertsEnabled() bool {
	raw, err := g.A.Store().Settings().Get(alertsEnabledKey)
	if err != nil {
		return true
	}
	return raw != "false"
}

func (g Gateway) SetAlertsEnabled(on bool) error {
	v := "false"
	if on { v = "true" }
	return g.A.Store().Settings().Set(alertsEnabledKey, v)
}
```

```go
func TestAlertRulesAndThrottle(t *testing.T) {
	now := time.Unix(1000, 0)
	a := newAlertState()
	if id, _, _ := a.evaluate(telemetry.RequestRecord{Status: 503, ErrorClass: "no_candidate"}, now); id != "no-candidate" {
		t.Fatalf("no_candidate id = %q", id)
	}
	if id, _, _ := a.evaluate(telemetry.RequestRecord{Status: 503, ErrorClass: "no_candidate"}, now.Add(time.Minute)); id != "" {
		t.Fatal("must throttle within 5m")
	}
	// 连续 3 次 FailoverCount>0 触发一次。
	for i := 0; i < 2; i++ {
		if id, _, _ := a.evaluate(telemetry.RequestRecord{FailoverCount: 1, Status: 200}, now.Add(time.Duration(i)*time.Second)); id != "" {
			t.Fatalf("premature failover alert %q", id)
		}
	}
	if id, _, _ := a.evaluate(telemetry.RequestRecord{FailoverCount: 1, Status: 200}, now.Add(3*time.Second)); id != "failover" {
		t.Fatalf("failover id = %q", id)
	}
	// 额度耗尽按渠道去重。
	q := telemetry.RequestRecord{Status: 429, ErrorClass: "quota_exhausted", Channel: "main"}
	if id, _, _ := a.evaluate(q, now.Add(10*time.Second)); id != "quota" {
		t.Fatalf("quota id = %q", id)
	}
	if id, _, _ := a.evaluate(q, now.Add(20*time.Second)); id != "" {
		t.Fatal("must throttle quota per channel")
	}
}
```

- [x] **Step 2: 实现 `alerts.go`**

```go
type alertState struct {
	mu          sync.Mutex
	lastNoCandidate time.Time
	lastFailover    time.Time
	consecutive     int
	lastQuota       map[string]time.Time
}

func newAlertState() *alertState { return &alertState{lastQuota: map[string]time.Time{}} }

func (a *alertState) evaluate(rec telemetry.RequestRecord, now time.Time) (string, string, string) {
	a.mu.Lock()
	defer a.mu.Unlock()
	if rec.FailoverCount > 0 {
		a.consecutive++
	} else {
		a.consecutive = 0
	}
	if rec.ErrorClass == "no_candidate" || (rec.Status == 503 && rec.ErrorClass == "no_candidate") {
		if now.Sub(a.lastNoCandidate) >= 5*time.Minute {
			a.lastNoCandidate = now
			return "no-candidate", "所有目标渠道冷却中", "AgentSwitch：当前所有目标都在冷却，请求可能失败。打开应用查看渠道/冷却原因。"
		}
	}
	if a.consecutive >= 3 && now.Sub(a.lastFailover) >= 5*time.Minute {
		a.lastFailover = now
		a.consecutive = 0
		return "failover", "连续故障转移", "AgentSwitch：最近多次请求发生目标切换，请检查主渠道与上游状态。"
	}
	if rec.ErrorClass == "quota_exhausted" {
		ch := rec.Channel
		if ch == "" { ch = "-" }
		if now.Sub(a.lastQuota[ch]) >= 10*time.Minute {
			a.lastQuota[ch] = now
			return "quota", "上游额度耗尽", "AgentSwitch：渠道「" + ch + "」返回额度耗尽，请更换 Key/渠道或调整目标链。"
		}
	}
	return "", "", ""
}
```

`desktop.go` 的 `OnRecord` 回调：

```go
a.OnRecord(func(rec telemetry.RequestRecord) {
	if u.Svc.Gateway.AlertsEnabled() {
		if id, title, body := u.alerts.evaluate(rec, time.Now()); id != "" {
			u.AlertMsg = title + "：" + body
			if mygo.NotificationsSupported() {
				_ = mygo.NewNotification(mygo.NotificationOptions{
					ID: "agentswitch-" + id, Title: title, Body: body, Group: "AgentSwitch",
				}).Show()
			}
		}
	}
	if w := win.Load(); w != nil {
		w.Update(func() {
			u.Recent = append([]telemetry.RequestRecord{rec}, u.Recent...)
			if len(u.Recent) > 100 { u.Recent = u.Recent[:100] }
		})
	}
})
```
